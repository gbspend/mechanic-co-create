# RUNNER: you run nonstop and bounce off walls; J turns you around. No gun.
# Jumping into a tile from below smashes it. Chasers patrol until you get
# close, then run at you and hop over walls. Lure them into pits.
# Vocabulary reference: the base RULES.txt (two folders up).

BIND jump: SPACE W UP Z K
BIND turn: J X

[player]
START -> FACE(RIGHT)
WALL_AHEAD -> TURN
BUTTON_PRESS(turn) -> TURN
AUTO -> WALK(3.2)

BUTTON_PRESS(jump) & ON_GROUND -> SET_YVEL(10)
AUTO -> ADD_YVEL(-0.6)
BUTTON_HELD(jump) & RISING -> ADD_YVEL(0.3)
AUTO -> CLAMP_YVEL(-12, 100)

# Head-butt: any tile we hit from underneath breaks
COLLIDE(t) & IS(t, _tile) & t(FROM_BELOW(self)) -> t(DIE)

FELL_OUT -> DIE

[chaser]
START -> FACE(LEFT)
AUTO -> ADD_YVEL(-0.6), CLAMP_YVEL(-12, 100)
LEDGE_AHEAD -> TURN
WALL_AHEAD -> TURN
AUTO -> WALK(1)
# Spotted you: face you and sprint. This overrides the TURNs above, so a
# chasing chaser runs off ledges, and hops when a wall is in its way.
NEAR(e, 260) & IS(e, player) -> FACE_TOWARD(e), WALK(2.6)
WALL_AHEAD & ON_GROUND -> ADD_YVEL(8)

COLLIDE(e) & IS(e, player) & e(FALLING) & FROM_ABOVE(e) -> DIE, e(SET_YVEL(9))
COLLIDE(e) & IS(e, player) -> e(DIE)
FELL_OUT -> DIE

[walker]
START -> FACE(LEFT)
AUTO -> ADD_YVEL(-0.6), CLAMP_YVEL(-12, 100)
WALL_AHEAD -> TURN
LEDGE_AHEAD -> TURN
AUTO -> WALK(1.2)

COLLIDE(e) & IS(e, player) & e(FALLING) & FROM_ABOVE(e) -> DIE, e(SET_YVEL(9))
COLLIDE(e) & IS(e, player) -> e(DIE)
FELL_OUT -> DIE

[flyer]
START -> ADD_YVEL(-2.8)
AUTO -> SPRING_Y(0.0016)

COLLIDE(e) & IS(e, player) & e(FALLING) & FROM_ABOVE(e) -> DIE, e(SET_YVEL(9))
COLLIDE(e) & IS(e, player) -> e(DIE)
