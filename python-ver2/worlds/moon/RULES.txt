# MOON: low gravity, a double jump, and a grenade launcher whose shots bounce.
# Walkers hop in place every so often; flyers swing much wider.
# Vocabulary reference: the base RULES.txt (two folders up).

BIND left:  LEFT A
BIND right: RIGHT D
BIND jump:  SPACE W UP Z K
BIND shoot: J X

[player]
AUTO -> SET_XVEL(0)
BUTTON_HELD(left)  -> ADD_XVEL(-3.5), FACE(LEFT)
BUTTON_HELD(right) -> ADD_XVEL(3.5), FACE(RIGHT)

# Double jump: a timer used as a flag. Landing makes the air jump READY again;
# using it sets a timer so long it never runs out mid-air.
ON_GROUND -> COOLDOWN(air_jump, 0)
BUTTON_PRESS(jump) & ON_GROUND -> SET_YVEL(7.5)
BUTTON_PRESS(jump) & IN_AIR & READY(air_jump) -> SET_YVEL(6.5), COOLDOWN(air_jump, 100000)

# Moon gravity (weaker still while holding jump), slow terminal velocity
AUTO -> ADD_YVEL(-0.25)
BUTTON_HELD(jump) & RISING -> ADD_YVEL(0.1)
AUTO -> CLAMP_YVEL(-6, 100)

BUTTON_PRESS(shoot) & READY(shoot) -> SPAWN(bullet, self), COOLDOWN(shoot, 25)

FELL_OUT -> DIE
COLLIDE(b) & IS(b, bullet) & b(OWNER(o)) & !IS(o, player) -> DIE, b(DIE)

[bullet]
# Grenade: lobbed, bounces along the ground, gone on walls or after 2 seconds
START -> ADD_YVEL(4)
AUTO -> ADD_YVEL(-0.2), WALK(6)
ON_GROUND -> SET_YVEL(3.5)
WALL_AHEAD -> DIE
AFTER(120) -> DIE

[walker]
START -> FACE(LEFT)
AUTO -> ADD_YVEL(-0.25), CLAMP_YVEL(-6, 100)
WALL_AHEAD -> TURN
LEDGE_AHEAD -> TURN
AUTO -> WALK(1)
# Hop straight up now and then (no sideways drift, so they don't hop off ledges).
# Random first wait, hop height and gap between hops keep them out of sync.
START -> COOLDOWN(hop, RANDOM(0, 90))
ON_GROUND & READY(hop) -> ADD_YVEL(RANDOM(4.5, 7)), COOLDOWN(hop, RANDOM(50, 110))
IN_AIR -> SET_XVEL(0)

COLLIDE(b) & IS(b, bullet) & b(OWNER(o)) & IS(o, player) -> DIE, b(DIE)
COLLIDE(e) & IS(e, player) & e(FALLING) & FROM_ABOVE(e) -> DIE, e(SET_YVEL(6))
COLLIDE(e) & IS(e, player) -> e(DIE)
FELL_OUT -> DIE

[flyer]
# Wider, slower swing: height = 3.2 / sqrt(0.0009) = ~107 px
START -> ADD_YVEL(-3.2)
AUTO -> SPRING_Y(0.0009)

COLLIDE(b) & IS(b, bullet) & b(OWNER(o)) & IS(o, player) -> DIE, b(DIE)
COLLIDE(e) & IS(e, player) & e(FALLING) & FROM_ABOVE(e) -> DIE, e(SET_YVEL(6))
COLLIDE(e) & IS(e, player) -> e(DIE)
